7)
function checkValue(value) {
  switch (value) {
    case 0:
      return "Number 0";
    case "0":
      return "String 0";
    case false:
      return "Boolean false";
    case null:
      return "null";
    case undefined:
      return "undefined";
    default:
      return "Unknown";
  }
}

console.log(checkValue(0));
console.log(checkValue("0"));
console.log(checkValue(false));
console.log(checkValue(null));
console.log(checkValue(undefined));
Output:
Number 0
String 0
Boolean false
null
undefined

switch checks with strict equality (===), so 0, "0" and false are all different cases. Same for null and undefined. With == they would match each other (0 == false, null == undefined), so the first matching case would catch them.

function calculate(a, b, operator) {
  switch (operator) {
    case "+":
      return a + b;
    case "-":
      return a - b;
    case "*":
      return a * b;
    case "/":
      if (b === 0) {
        return "Cannot divide by zero";
      }
      return a / b;
    case "%":
      if (b === 0) {
        return "Cannot divide by zero";
      }
      return a % b;
    case "**":
      return a ** b;
    default:
      return "Invalid operator";
  }
}

console.log(calculate(10, 5, "+"));
console.log(calculate(10, 5, "-"));
console.log(calculate(10, 5, "*"));
console.log(calculate(10, 5, "/"));
console.log(calculate(10, 3, "%"));
console.log(calculate(2, 5, "**"));
console.log(calculate(10, 0, "/"));

15
5
50
2
1
32
